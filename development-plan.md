## Technology Decisions & Rationale

### Data Model: Hybrid Relational + JSONB (Suggestion 3) with Graph Extension Path (Suggestion 4)

**Decision:** Start with Data Model Suggestion 3 (Hybrid Relational + JSONB) for Phases 1-6, and introduce the graph layer from Suggestion 4 in Phase 7+.

**Rationale:**
- Suggestion 1 (pure normalized) imposes too many schema migrations for a domain where signal types, NLP outputs, and source connectors evolve rapidly. 31+ tables from day one is excessive pre-optimization.
- Suggestion 2 (event-sourced CQRS) adds cognitive and infrastructure overhead that slows early delivery. Event sourcing can be retrofitted into the audit layer later if audit requirements demand it.
- Suggestion 3 (hybrid) delivers the best velocity-to-safety ratio: relational foreign keys for structural integrity (org, user, company, signal relationships) combined with JSONB for variable NLP payloads, source-specific configs, and per-org custom fields. ~15 tables at MVP. PostgreSQL JSONB with GIN indexes is battle-tested at this scale (Notion, Linear use the same pattern).
- Suggestion 4 (graph) becomes critical for cross-signal synthesis and relationship traversal ("Company A hired from Company B, which just filed a patent overlapping Company C's product"). However, graph complexity is not needed until the NLP pipeline and entity resolution are mature. The PostgreSQL-native graph (graph_nodes / graph_edges with recursive CTEs) avoids premature Neo4j infrastructure cost while enabling advanced analysis later.

```python
# src/mia/auth/oidc.py
from authlib.integrations.starlette_client import OAuth
from fastapi import Depends, HTTPException, Security
from fastapi.security import HTTPBearer, APIKeyHeader

security = HTTPBearer(auto_error=False)
api_key_header = APIKeyHeader(name="X-API-Key", auto_error=False)

async def get_current_user(
    token: str = Security(security),
    api_key: str = Security(api_key_header),
    db: AsyncSession = Depends(get_db),
) -> User:
    """Authenticate via Bearer token (OIDC) or API key."""
    if token:
        return await authenticate_oidc_token(token.credentials, db)
    elif api_key:
        return await authenticate_api_key(api_key, db)
    raise HTTPException(status_code=401, detail="Authentication required")

# src/mia/auth/rbac.py
from functools import wraps

class Permission:
    SIGNAL_READ = "signal:read"
    SIGNAL_WRITE = "signal:write"
    COMPANY_MANAGE = "company:manage"
    BATTLECARD_MANAGE = "battlecard:manage"
    SOURCE_MANAGE = "source:manage"
    ORG_ADMIN = "org:admin"

ROLE_PERMISSIONS = {
    "viewer": {Permission.SIGNAL_READ},
    "member": {Permission.SIGNAL_READ, Permission.SIGNAL_WRITE, Permission.COMPANY_MANAGE,
               Permission.BATTLECARD_MANAGE},
    "admin": {Permission.SIGNAL_READ, Permission.SIGNAL_WRITE, Permission.COMPANY_MANAGE,
              Permission.BATTLECARD_MANAGE, Permission.SOURCE_MANAGE, Permission.ORG_ADMIN},
}

def require_permission(permission: str):
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, user: User = Depends(get_current_user), **kwargs):
            if permission not in ROLE_PERMISSIONS.get(user.role, set()):
                raise HTTPException(status_code=403, detail="Insufficient permissions")
            return await func(*args, user=user, **kwargs)
        return wrapper
    return decorator
```

### Task 2.1: RSS/Atom Feed Crawler

**What:** Implement an RSS 2.0 and Atom 1.0 feed crawler that fetches, parses, and stores articles. Support OPML import for bulk feed onboarding. Track fetch history and health in the `sources` table.

**Design:**
```python
# src/mia/crawlers/rss_crawler.py
import feedparser
from datetime import datetime, timezone

class RSSCrawler:
    """Crawls RSS 2.0 and Atom 1.0 feeds."""

    async def crawl(self, source: Source, db: AsyncSession) -> CrawlResult:
        # 1. Check robots.txt compliance
        if not await self._check_robots_txt(source.url):
            return CrawlResult(status="robots_blocked", items_found=0, items_new=0)

        # 2. Fetch feed with appropriate User-Agent
        response = await self._fetch_feed(source.url)

        # 3. Parse with feedparser (handles both RSS 2.0 and Atom 1.0)
        feed = feedparser.parse(response.text)

        # 4. Extract articles, applying Schema.org metadata where available
        articles = []
        for entry in feed.entries:
            content_hash = self._compute_hash(entry)
            if await self._is_duplicate(content_hash, db):
                continue
```

```python
from playwright.async_api import async_playwright
from urllib.robotparser import RobotFileParser

class WebsiteCrawler:
    """Crawls JavaScript-rendered web pages via Playwright."""

    async def crawl(self, source: Source, db: AsyncSession) -> CrawlResult:
        # 1. Check and cache robots.txt

            await browser.close()

        # 3. Extract content using configured selectors
        selectors = source.config.get("selectors", {})
        articles = self._extract_articles(html, selectors, source)

        # 4. Check for changes via content hash
        new_articles = []
        for article in articles:
            if not await self._is_duplicate(article.content_hash, db):
                new_articles.append(article)

        db.add_all(new_articles)
        await db.commit()
        return CrawlResult(status="success", items_found=len(articles), items_new=len(new_articles))
```

```python
import spacy
from spacy.tokens import Doc

class NERProcessor:
    """Extracts named entities from article text and resolves to tracked entities."""

    def __init__(self, model_name: str = "en_core_web_trf"):
        self.nlp = spacy.load(model_name)

    async def extract_entities(
        self, text: str, organization_id: str, db: AsyncSession
    ) -> list[ExtractedEntity]:
        doc = self.nlp(text)
        entities = []

        for ent in doc.ents:
            if ent.label_ in ("ORG", "PERSON", "PRODUCT", "GPE", "MONEY"):
                resolved = await self._resolve_entity(ent, organization_id, db)
                entities.append(ExtractedEntity(
                    type=self._map_spacy_label(ent.label_),
                    value=ent.text,
                    normalized=resolved.canonical_name if resolved else ent.text.lower(),
                    resolved_id=resolved.id if resolved else None,
                    confidence=self._compute_confidence(ent, resolved),
                    start_offset=ent.start_char,
                    end_offset=ent.end_char,
                ))

        return entities

    async def _resolve_entity(self, ent, org_id, db) -> Company | Person | None:
        """Resolve an NER span to a tracked company or person."""
        if ent.label_ == "ORG":
            # Check exact match on canonical_name
            result = await db.execute(
                select(Company)
                .where(Company.organization_id == org_id)
                .where(Company.canonical_name == ent.text.lower().strip())
            )
            company = result.scalar_one_or_none()
            if company:
                return company

            # Check aliases in JSONB details
            result = await db.execute(
                select(Company)
                .where(Company.organization_id == org_id)
                .where(Company.details["aliases"].astext.ilike(f"%{ent.text}%"))
            )
            return result.scalar_one_or_none()
        return None
```

```python
# src/mia/nlp/sentiment.py
from transformers import AutoModelForSequenceClassification, AutoTokenizer
import torch

class SentimentAnalyzer:
    """Financial/business sentiment analysis using FinBERT."""

    def __init__(self, model_name: str = "ProsusAI/finbert"):
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.model = AutoModelForSequenceClassification.from_pretrained(model_name)
        self.labels = ["positive", "negative", "neutral"]

    def analyze(self, text: str) -> SentimentResult:
        inputs = self.tokenizer(text, return_tensors="pt", truncation=True, max_length=512)
        with torch.no_grad():
            outputs = self.model(**inputs)
        probabilities = torch.softmax(outputs.logits, dim=1)[0]

        scores = {label: float(prob) for label, prob in zip(self.labels, probabilities)}
        dominant = max(scores, key=scores.get)

        # Compute continuous score: positive=+1, negative=-1, neutral=0
        continuous_score = scores["positive"] - scores["negative"]

        # Detect mixed sentiment
        if scores["positive"] > 0.3 and scores["negative"] > 0.3:
            dominant = "mixed"

        return SentimentResult(
            sentiment=dominant,
            sentiment_score=round(continuous_score, 3),
            scores=scores,
        )
```

```python
class NLPPipeline:
    """Orchestrates NER, classification, and sentiment analysis to produce signals."""

    def __init__(self):
        self.ner = NERProcessor()
        self.classifier = SignalClassifier()
        self.sentiment = SentimentAnalyzer()

        if not company_entities:

                company_id=entity.resolved_id if entity else None,

                signal_category=classification.signal_category,

                    "sentiment": sentiment.sentiment,

                    "confidence": classification.confidence,

                    "source_url": article.url,
                    "published_at": article.published_at.isoformat() if article.published_at else None,
                    "entities": [e.to_dict() for e in entities],
                    "topics": [],
                    "nlp_model_version": "pipeline-v1.0",
                },
            )
            signals.append(signal)

        db.add_all(signals)
        await db.commit()

        # 6. Update company signal counts
        for signal in signals:
            if signal.company_id:
                await self._increment_company_signal_count(signal.company_id, db)
```

```python
import hmac, hashlib, time
import httpx

class WebhookDispatcher:
    """Delivers webhook events per the Standard Webhooks specification."""

    async def dispatch(self, event_type: str, payload: dict, organization_id: str, db: AsyncSession):
        endpoints = await db.execute(
            select(Webhook)
            .where(Webhook.organization_id == organization_id)
            .where(Webhook.is_active == True)
            .where(Webhook.config["event_types"].contains(f'["{event_type}"]'))
        )

        for endpoint in endpoints.scalars():
            await self._deliver(endpoint, event_type, payload)

    async def _deliver(self, endpoint: Webhook, event_type: str, payload: dict):
        timestamp = str(int(time.time()))
        webhook_id = str(uuid.uuid4())
        body = json.dumps(payload)

        # Standard Webhooks signature: v1,<base64(hmac-sha256(secret, "webhook_id.timestamp.body"))>
        signature_input = f"{webhook_id}.{timestamp}.{body}"
        signature = hmac.new(
            endpoint.secret_hash.encode(),
            signature_input.encode(),
            hashlib.sha256,
        ).digest()
        signature_b64 = f"v1,{base64.b64encode(signature).decode()}"

        headers = {
            "webhook-id": webhook_id,
            "webhook-timestamp": timestamp,
            "webhook-signature": signature_b64,
            "content-type": "application/json",
        }

        async with httpx.AsyncClient() as client:
            response = await client.post(endpoint.url, content=body, headers=headers, timeout=30)
```

```python
class AlertDispatcher:
    """Evaluates new signals against subscriptions and dispatches alerts."""

    async def evaluate_signal(self, signal: Signal, db: AsyncSession):
        subscriptions = await db.execute(
            select(AlertSubscription)
            .where(AlertSubscription.organization_id == signal.organization_id)
            .where(AlertSubscription.is_active == True)
        )

        for sub in subscriptions.scalars():
            if self._matches_filters(signal, sub.filters):
                if sub.delivery_frequency == "realtime":
                    await self._deliver_immediately(signal, sub)
                else:
                    await self._queue_for_digest(signal, sub)

    def _matches_filters(self, signal: Signal, filters: dict) -> bool:
        if filters.get("companies") and str(signal.company_id) not in filters["companies"]:
            return False
        if filters.get("signal_types") and signal.signal_type not in filters["signal_types"]:
            return False
        if filters.get("categories") and signal.signal_category not in filters["categories"]:
            return False
        if filters.get("min_relevance"):
            score = signal.payload.get("relevance_score", 0)
            if score < filters["min_relevance"]:
                return False
        return True
```

```python
from dataclasses import dataclass

@dataclass
class SignalPattern:
    name: str
    description: str
    required_categories: list[str]
    min_signals_per_category: dict[str, int]
    time_window_days: int
    confidence_threshold: float

PREDEFINED_PATTERNS = [
    SignalPattern(
        name="strategic_pivot",
        description="Company appears to be pivoting strategy",
        required_categories=["talent", "product"],
        min_signals_per_category={"talent": 2, "product": 1},
        time_window_days=90,
        confidence_threshold=0.7,
    ),
    SignalPattern(
        name="aggressive_pricing",
        description="Company adopting aggressive pricing strategy",
        required_categories=["pricing", "talent"],
        min_signals_per_category={"pricing": 1, "talent": 1},
        time_window_days=60,
        confidence_threshold=0.6,
    ),
    SignalPattern(
        name="market_expansion",
        description="Company expanding into new markets",
        required_categories=["market", "partnership", "talent"],
        min_signals_per_category={"market": 1, "partnership": 1},
        time_window_days=90,
        confidence_threshold=0.6,
    ),
    SignalPattern(
        name="acquisition_preparation",
        description="Company showing pre-acquisition signals",
        required_categories=["funding", "talent"],
        min_signals_per_category={"funding": 1, "talent": 2},
        time_window_days=120,
        confidence_threshold=0.7,
    ),
]

class PatternDetector:
    async def detect_patterns(self, company_id: str, db: AsyncSession) -> list[DetectedPattern]:
        detected = []
        for pattern in PREDEFINED_PATTERNS:
            signals_by_category = await self._get_signals_by_category(
                company_id, pattern.required_categories, pattern.time_window_days, db
            )

            meets_threshold = all(
                len(signals_by_category.get(cat, [])) >= pattern.min_signals_per_category.get(cat, 1)
                for cat in pattern.required_categories
            )

            if meets_threshold:
                all_signals = [s for sigs in signals_by_category.values() for s in sigs]
                confidence = self._compute_pattern_confidence(all_signals, pattern)
                if confidence >= pattern.confidence_threshold:
                    detected.append(DetectedPattern(
                        pattern=pattern,
                        signals=all_signals,
                        confidence=confidence,
                    ))

        return detected
```

**Design:**

```python
# src/mia/services/battlecard_engine.py
class BattlecardEngine:
    async def ai_draft_section(
        self, battlecard_id: str, section_type: str, db: AsyncSession
    ) -> dict:
        battlecard = await db.get(Battlecard, battlecard_id)
        competitor_id = battlecard.competitor_id

        # Gather evidence: recent signals, narratives, win/loss records
        signals = await self._get_recent_signals(competitor_id, days=90, db=db)
        narratives = await self._get_narratives(competitor_id, db=db)
        win_loss = await self._get_win_loss_records(competitor_id, db=db)

        prompt = self._build_section_prompt(section_type, signals, narratives, win_loss)

        response = self.client.messages.create(
            model="claude-sonnet-4-20250514",
            max_tokens=2000,
            messages=[{"role": "user", "content": prompt}],
        )

        # Update the section in the JSONB array
        sections = battlecard.sections.copy()
        for section in sections:
            if section["section_type"] == section_type:
                section["content"] = response.content[0].text
                section["ai_generated"] = True
                section["source_signal_ids"] = [str(s.id) for s in signals[:20]]
                section["last_verified_at"] = None
                break
        battlecard.sections = sections
        await db.commit()

        return section
```
